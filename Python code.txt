class userdetails(db.Model):
    id = db.Column(db.Integer, primary_key=True, autoincrement=True)  # Add autoincrement explicitly
    name = db.Column(db.String(100), nullable=False)
    email = db.Column(db.String(255), unique=True, nullable=False)
    password = db.Column(db.String(255), nullable=False)
    mobilenumber = db.Column(db.String(15), nullable=False)
    location = db.Column(db.String(100), nullable=False)

@app.route('/signup', methods=['POST'])
def signup():
    try:
        data = request.get_json()
        name = data.get('name')
        email = data.get('email')
        password = data.get('password')
        mobilenumber = data.get('mobilenumber')
        location = data.get('location')

        if not all([name, email, password, mobilenumber, location]):
            return jsonify({'message': 'All fields are required'}), 400

        # Check if user already exists
        if userdetails.query.filter_by(email=email).first():
            return jsonify({'message': 'User already exists'}), 409

        password_hash = bcrypt.generate_password_hash(password).decode('utf-8')

        new_user = userdetails(
            name=name,
            email=email,
            password=password_hash,
            mobilenumber=mobilenumber,
            location=location
        )

        db.session.add(new_user)
        db.session.commit()

        return jsonify({'message': 'User registered successfully'}), 201

    except Exception as e:
        db.session.rollback()  # Add rollback on error
        print(f"Error: {str(e)}")  # For debugging
        return jsonify({'error': str(e)}), 500


@app.route('/login', methods=['POST'])
def login():
    try:
        data = request.get_json()
        email = data.get('email')
        password = data.get('password')

        if not all([email, password]):
            return jsonify({'message': 'Email and password are required'}), 400

        user = userdetails.query.filter_by(email=email).first()
        if not user or not bcrypt.check_password_hash(user.password, password):
            return jsonify({'message': 'Invalid email or password'}), 401

        return jsonify({
                'message': 'Login successful',
                'user': {
                    'id': user.id,
                    'name': user.name,
                    'email': user.email,
                    'mobilenumber': user.mobilenumber,
                    'location': user.location
                }
            }), 200

    except Exception as e:
        return jsonify({'error': str(e)}), 500

@app.route('/evaluate', methods=['GET'])
def get_all_evaluates():
    try:

        query = text("""
            SELECT
                e.id,
                e.name,
                e.location,
                e.type,
                e.year,
                e.km,
                e.fuel,
                e.price,
                e.seller_email,
                u.name AS seller_name,
                u.mobilenumber AS seller_mobile
            FROM evaluate e
            LEFT JOIN userdetails u
            ON e.seller_email = u.email
        """)

        result = db.session.execute(query)

        cars = []

        for row in result:
            cars.append({
                'id': row.id,
                'name': row.name,
                'location': row.location,
                'type': row.type,
                'year': row.year,
                'km': row.km,
                'fuel': row.fuel,
                'price': row.price,
                'seller_email': row.seller_email,
                'seller_name': row.seller_name,
                'seller_mobile': row.seller_mobile
            })

        return jsonify(cars), 200

    except Exception as e:
        return jsonify({'error': str(e)}), 500
